# Cobalt's patch series, in application order.
#
# This file is the answer to "how do I reproduce your APK". Every change Cobalt
# makes to Chromium is listed here, and tools/patches/apply-all.sh runs the list
# top to bottom. Nothing else is applied by hand -- if a change is not in this
# file, it does not exist as far as a fresh checkout is concerned.
#
# Format, one entry per line:
#
#   <kind> <argument...>
#
#   kiwi   <patch-name>       apply patches/kiwi-105/<name>.patch via apply-batch.sh
#   tool   <script> [args]    run a script from tools/, relative to the repo root
#   backport <repo> <CVE>     apply patches/security/<CVE>.diff inside <repo>
#                             (a path under $SRC, "." for src) -- all or nothing
#   note   <text>             printed, does nothing -- section headings
#
# $SRC expands to the Chromium checkout. Every tool must be idempotent: the
# series is re-run after every gclient sync, and a second run must be a no-op.

note ===== Batch 1: extension core plumbing (docs/batch-1-triage.md) =====

# The eight patches that apply as-is. extension_protocols.cc is deliberately
# absent: it disabled AllowExtensionResourceLoad, and apply-batch.sh now refuses
# the whole class. See docs/decisions/0011-refuse-mechanical-disablers.md.
kiwi extensions_common_url_pattern.h
# The line this adds is rewritten later by cobalt-chromesearch-scheme.py, so
# a reverse-apply check cannot recognise it. Match on the mask instead.
kiwi extensions_common_url_pattern.cc :: SCHEME_CHROMESEARCH
kiwi extensions_common_extension.cc
kiwi extensions_common_manifest_handlers_permissions_parser.cc
kiwi extensions_common_manifest_handlers_csp_info.cc
# Applied with fuzz, so the reverse check does not match it either.
kiwi extensions_common_user_script.cc :: SCHEME_CHROMESEARCH

note ===== Cobalt's own ports and rewrites =====

# Kiwi's "|| true" platform hacks, re-expressed as real Android conditions.
tool patches/cobalt-android-conditions.py --src $SRC

# content::kChromeSearchScheme moved up a layer between M105 and M140.
tool patches/cobalt-chromesearch-scheme.py $SRC

# MV2 enabled by default. Decision 0005 -- never remove this.
tool patches/cobalt-mv2-defaults.py --src $SRC

# MV2's browserAction/pageAction schemas, which upstream ships only when
# enable_extensions is set. Without them the renderer aborts on any MV2
# extension with a browser_action -- uBlock Origin included.
tool patches/cobalt-mv2-action-schemas.py $SRC

# webNavigation, which upstream marks as not supported on desktop_android.
# uBlock Origin cannot start without it. The tab-strip half stays desktop-only.
tool patches/cobalt-webnavigation-android.py $SRC

note ===== Search engines =====

# WARNING: this one writes into third_party/search_engines_data/resources,

tool patches/cobalt-search-engines.py $SRC

note ===== Branding =====

tool assets/brand-chromium.py --src $SRC
tool assets/brand-android-logos.py --src $SRC

# The launcher icon is three layers, not one. brand-chromium.py replaces the
# square foreground; these are the round icon's background and the monochrome
# themed icon, both of which were still Chromium's pinwheel.
tool assets/brand-launcher-layers.py --src $SRC

# brand-chromium.py writes the same PNG to app_icon.png and layered_app_icon.png.
# Android lint refuses byte-identical icons and its step fails on any output at
# all, so this broke chrome_public_apk. It is also simply wrong: the layered one
# is an adaptive icon's foreground and belongs inset in the 66/108 safe zone,
# or the launcher's mask clips it.
tool assets/inset-adaptive-foreground.py --src $SRC
tool assets/brand-strings.py --src $SRC

# base::FileEnumerator never handled /SAF/ virtual document paths, which is

tool patches/cobalt-saf-file-enumerator.py $SRC

note ===== Features Cobalt does not ship =====

# args.gn's enable_vr/openxr/arcore/cardboard remove WebXR's device backends
# but leave navigator.xr exposed and inline sessions working -- measured on a
# device, not read off the GN files. This removes the APIs themselves: WebXR,
# WebUSB and Web NFC. WebHID needs no patch (upstream leaves it off on Android)
# and Web Bluetooth stays, already ASK-by-default. See docs/device-apis.md.
tool patches/cobalt-disable-device-apis.py $SRC

note ===== Google Play Services removal (decision 0013) =====

# Barcode and text detection on Android are GMS vision wrappers with no other
# implementation; face detection has an AOSP one. This takes the vision AARs
# out, leaving upstream's own no-Play-Services behaviour. Removes 4 of 18 GMS
# modules from the APK. See docs/gms-removal.md.
tool patches/cobalt-gms-shape-detection.py $SRC

# Geolocation keeps working through android.location.LocationManager, which is
# what upstream already falls back to without Play Services. Drops the fused
# provider. Note Chromium never runs the googleapis.com/geolocation network
# path on Android -- it is in the non-Android branch of the BUILD.gn.
tool patches/cobalt-gms-geolocation.py $SRC

# chrome_java listed eight GMS modules and imports three. The other five were
# removed and chrome_java rebuilt to prove it -- a Java target can legitimately
# need a dep it never imports. Edges 43 -> 38, no Java changes.
tool patches/cobalt-gms-chrome-java-dead-deps.py $SRC

# ExternalAuthUtils is the question the rest of the tree asks before doing
# anything GMS-shaped -- 14 non-test call sites. Answering false unconditionally
# makes all of them take upstream's own no-Play-Services path before their own
# removals are written, which makes the rest of 0013 safer.
tool patches/cobalt-gms-externalauth.py $SRC

# Four one-edge targets, none touching chrome/android: language and
# module_installer listed tasks and never used it, omaha's only use was a string
# constant, and webauthn's Fido2ApiCall block was already unreachable because
# externalauth above it now always answers no.
tool patches/cobalt-gms-singles.py $SRC

# WebOTP. Not a feature removal so much as a startup fix: SmsProviderGms asks
# GoogleApiAvailability whether Play Services is present, and on an app with no
# GMS manifest tag that question throws rather than answering, killing the
# browser process during startup. Cobalt answers it without asking.
tool patches/cobalt-gms-webotp.py $SRC

note ===== Shell =====

# ContentViewRenderView, the Android View a WebContents composites into. Chrome
# uses chrome/android's CompositorViewHolder instead, so neither half of this is
# in chrome_public_apk's graph. Both halves go in: the Java into chrome_all_java
# (which is where GEN_JNI is generated from), the C++ into libchrome.
tool patches/cobalt-content-view-render-view.py $SRC

# Compiling that file into Chrome's build also lints it for the first time,
# and it fails NewApi five times on calls guarded by a native feature check.
# content_shell_apk, the only target that ever built it, does not enable lint.
tool patches/cobalt-lint-content-view-render-view.py $SRC

# A WebContentsDelegate for tabs Chrome's TabAndroid does not own. Without one,
# Chromium refuses every download (DownloadRequestLimiter treats a missing
# delegate as "no") and has nowhere to ask about new windows or fullscreen.
tool patches/cobalt-webcontents-delegate.py $SRC


# Chromium cannot compile Compose (its kotlinc has no plugin support), so
# Cobalt's interface is built in the Gradle project and Chromium arrives as an
# AAR. This adds the dist_aar target. It is not in chrome_public_apk's graph, so
# it costs the normal build nothing. See docs/shell-integration.md.
tool patches/cobalt-content-aar.py $SRC

note ===== Bundled extensions =====

# Stages the signed uBO CRX into the APK and configures the policy that makes
# it disableable but not removable. Decision 0006.
tool patches/cobalt-bundle-ublock.py $SRC

# Android kills background renderers; Chromium terminates the extension and
# waits for a crash bubble Cobalt does not have, so a bundled content blocker
# silently stops blocking for the rest of the session. Reproduced with
# kill -9 on the extension renderer. Reloads with backoff and a cap.
tool patches/cobalt-extension-auto-reload.py $SRC
